using UnityEngine; public class PlayerController : MonoBehaviour { public float moveSpeed = 5f; public float attackRange = 2.5f; public int attackDamage = 20; public LayerMask interactableLayer; void Update() { // Harakatlanish float h = Input.GetAxis("Horizontal"); float v = Input.GetAxis("Vertical"); Vector3 move = new Vector3(h, 0, v).normalized; transform.Translate(move * moveSpeed * Time.deltaTime, Space.World); if (move != Vector3.zero) transform.forward = move; // Zarba berish (Daraxt kesish yoki Monstrga hujum) if (Input.GetMouseButtonDown(0)) { Attack(); } } void Attack() { Collider[] hits = Physics.OverlapSphere(transform.position + transform.forward, attackRange, interactableLayer); foreach (Collider hit in hits) { // Agar daraxt yoki tosh bo'lsa ResourceNode resource = hit.GetComponent(); if (resource != null) { resource.Gather(attackDamage); } // Agar monstr bo'lsa WildMonster monster = hit.GetComponent(); if (monster != null) { monster.TakeDamage(attackDamage); } } } }
